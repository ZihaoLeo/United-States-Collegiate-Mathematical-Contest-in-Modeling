function createfigure(X1, Y1)

%  figure
figure1 = figure('WindowState','maximized');

%  axes
axes1 = axes('Parent',figure1);
hold(axes1,'on');

% plot
plot(X1,Y1,'MarkerSize',3,'Marker','.','LineStyle','none','Color',[0 0 1]);

%  ylabel
ylabel('latitude');

%  xlabel
xlabel('longitude');

%  title
title('Brightkite user sign in area distribution');
%
 xlim(axes1,[-180 180]);
 ylim(axes1,[-90 90]);
box(axes1,'on');
end

function createfigure3(X3)
a=0;
b=0;
c=0;
d=0;
t=0;

for w=1:1:4747287
    X=char(X3(w));
    if(X~=' ')
    if X(1:4)=='2009'
        t=t+1;
        switch X(6:7)
            case '01'
                a=a+1;
            case '02'
                a=a+1;
            case '03'
                a=a+1;
            case '04'
                b=b+1;
            case '05'
                b=b+1;
            case '06'
                b=b+1;
            case '07'
                c=c+1;
            case '08'
                c=c+1;
            case '09'
                c=c+1;
            case '10'
                d=d+1;
            case '11'
                d=d+1;
            case '12'
                d=d+1;
        end
    end
    end
end
Y3=[a,b,c,d];
bar(Y3,'Barwidth',0.4);
ylabel('number');
xlabel('quarter');
title('Distribution of check-in time in Brightkite( quarter)');

end
